"ล่ามแปลภาษาและบริษัทขนส่งพัสดุข้ามชาติ" 📦🌐
ลองจินตนาการว่าบริษัทของคุณต้องการส่งเอกสารหรือสั่งของจากคู่ค้าต่างประเทศ:

  • HttpClientFactory (กองยานพาหนะส่วนกลาง): เหมือนบริษัทโลจิสติกส์ที่มีอู่รถส่วนกลาง จัดการบำรุงรักษารถบรรทุกและคนขับ ไม่ต้องซื้อรถคันใหม่ทุกครั้งที่ส่งจดหมาย (ไม่สร้าง connection ใหม่จน socket หมด) และเมื่อเส้นทางเปลี่ยนก็สามารถปรับเปลี่ยนเส้นทางได้ทันที (DNS refresh)
  • System.Text.Json (ล่ามแปลภาษาและมาตรฐานกล่องพัสดุ): ก่อนส่งของ เราต้องแพ็กสินค้าลงกล่องตามมาตรฐานสากล (JSON Serialization) และเมื่อได้รับกล่องพัสดุมา ล่ามก็แปลและแกะของออกมาเป็นวัตถุที่เราใช้งานได้ทันที (Deserialization)
  • Polly Resilience (ประกันภัยและแผนรับมือเหตุฉุกเฉิน): หากถนนปิดหรือฝนตกหนัก ขบวนรถจะรอสักครู่แล้วลองเดินทางใหม่ (Retry with Exponential Backoff) และถ้าสะพานขาดถาวร จะสั่งปิดถนนชั่วคราวไม่ให้รถคันอื่นขับไปติดขัดเปล่าๆ (Circuit Breaker)
  • Webhooks & HMAC Signature (ระบบแจ้งเตือนพัสดุพร้อมตราประทับลับ): เมื่อคู่ค้าทำงานเสร็จ เขาไม่ต้องรอเราโทรไปถามซ้ำๆ แต่จะส่งรถมาเคาะประตูบ้านเราทันที (Webhook Callback) โดยมีตราประทับครั่งลับพิเศษ (HMAC-SHA256 Signature) เพื่อให้เรามั่นใจ 100% ว่าไม่ใช่พัสดุปลอมจากมิจฉาชีพ!
01

HttpClient vs IHttpClientFactory: ปัญหาที่แท้จริงเบื้องหลัง HTTP Connection

ทำไมการเขียน using var client = new HttpClient() ถึงเป็นกับดักทำลาย Production Server และทำไม Singleton HttpClient ก็ไม่ใช่คำตอบที่สมบูรณ์

นักพัฒนา C# มือใหม่เกือบทุกคนเริ่มต้นด้วยการมองว่า HttpClient สืบทอดมาจาก IDisposable ดังนั้นจึงเขียนครอบด้วยบล็อก using เหมือนกับการเปิดไฟล์หรือเชื่อมต่อ Database แต่ในระดับเครือข่าย OS ระดับต่ำ (TCP Sockets) พฤติกรรมของมันกลับต่างออกไปอย่างสิ้นเชิง!

❌ ปัญหาที่ 1: Socket Exhaustion (เมื่อ new HttpClient บ่อยๆ)

เมื่อสั่ง Dispose() ตัว HttpClient ตัว Socket TCP เบื้องล่างจะไม่ถูกปิดในทันที แต่ระบบปฏิบัติการ Windows/Linux จะค้างสถานะไว้ในโหมด TIME_WAIT เป็นเวลา 2-4 นาที เพื่อรอ packet ที่อาจจะตกค้างในเครือข่าย

หากเซิร์ฟเวอร์ของคุณมี Request เข้ามา 500-1,000 ครั้ง/วินาที พอร์ตเชื่อมต่อ (Outbound Sockets) จะถูกใช้จนเต็มหมดเกลี้ยง ส่งผลให้เกิดข้อผิดพลาดรุนแรง:

System.Net.Sockets.SocketException: Only one usage of each socket address (protocol/network address/port) is normally permitted.

⚠️ ปัญหาที่ 2: DNS Refresh Issue (เมื่อใช้ Singleton HttpClient)

หลายคนแก้ปัญหา Socket Exhaustion ด้วยการประกาศ static readonly HttpClient ไว้ตัวเดียวตลอดอายุของแอปพลิเคชัน แต่วิธีนี้ทำให้เกิดปัญหาใหม่!

HttpClient แบบ Singleton จะเปิด TCP connection แช่ค้างไว้และไม่ยอมทำ DNS Lookup ซ้ำ หาก Third-party API หรือ Cloud Service ปลายทาง (เช่น Azure, AWS, Cloudflare) มีการอัปเดต IP หรือ Failover ไปยังเซิร์ฟเวอร์ใหม่ แอปของคุณจะยังยิงไปที่ IP เก่าจน Request พังทั้งหมด (Downtime)

💡 ทางออกระดับสากล: IHttpClientFactory ใน .NET Core / .NET 8
.NET แนะนำให้ใช้ IHttpClientFactory ซึ่งทำหน้าที่แยก Object ของ HttpClient ออกจาก HttpMessageHandler โดย Handler จะถูกบริหารผ่าน Connection Pool ที่มีอายุขัย (Default 2 นาที) เมื่อ Handler หมดอายุ Connection ใหม่จะทำการ Re-resolve DNS เสมอ และ Socket จะถูกนำกลับมาใช้ซ้ำได้อย่างมีประสิทธิภาพสูงสุด ป้องกันทั้ง Socket Exhaustion และ DNS Stale ได้แบบ 100%!

02

การลงทะเบียน Typed Client และ Named Client ใน Program.cs

เรียนรู้รูปแบบ Best Practice ทั้ง 3 วิธีในการลงทะเบียนและใช้งาน HttpClient ใน ASP.NET Core

ใน .NET มีรูปแบบการใช้งาน IHttpClientFactory หลักๆ 3 รูปแบบ โดยรูปแบบที่แนะนำที่สุดสำหรับงานระดับ Enterprise คือ Typed Client

Typed Client รวมการตั้งค่า URL, Header, Authorization และตรรกะการเรียก API เข้าไว้ในคลาส Service เฉพาะตัว ทำให้โค้ดสะอาด มี Type-Safety และง่ายต่อการทำ Unit Test

📄 Services/PaymentClient.cs (การสร้าง Typed Client Class) 
using System.Net.Http.Json;

namespace DotNetHub.Integration.Services;

// DTO Models
public record PaymentRequest(string OrderId, decimal Amount, string Currency);
public record PaymentResponse(string TransactionId, string Status, DateTime ProcessedAt);

public interface IPaymentClient
{
    Task ProcessPaymentAsync(PaymentRequest request, CancellationToken ct = default);
}

// Typed Client: รับ HttpClient ผ่าน Dependency Injection โดยตรง
public class PaymentClient : IPaymentClient
{
    private readonly HttpClient _httpClient;
    private readonly ILogger _logger;

    public PaymentClient(HttpClient httpClient, ILogger logger)
    {
        _httpClient = httpClient;
        _logger = logger;
    }

    public async Task ProcessPaymentAsync(PaymentRequest request, CancellationToken ct = default)
    {
        _logger.LogInformation("กำลังส่งคำขอชำระเงินสำหรับ Order: {OrderId}", request.OrderId);

        var response = await _httpClient.PostAsJsonAsync("/api/v1/charges", request, ct);
        
        // ตรวจสอบว่า HTTP Status สำเร็จหรือไม่ (200-299)
        response.EnsureSuccessStatusCode();

        return await response.Content.ReadFromJsonAsync(cancellationToken: ct);
    }
}
⚙️ Program.cs (การลงทะเบียน Typed Client ลง DI Container) 
var builder = WebApplication.CreateBuilder(args);

// ลงทะเบียน Typed Client พร้อมตั้งค่า BaseAddress และ Headers ส่วนกลาง
builder.Services.AddHttpClient((serviceProvider, client) =>
{
    var config = serviceProvider.GetRequiredService();
    
    client.BaseAddress = new Uri(config["PaymentGateway:BaseUrl"] ?? "https://api.paymenthub.com");
    client.DefaultRequestHeaders.Add("Accept", "application/json");
    client.DefaultRequestHeaders.Add("X-Api-Key", config["PaymentGateway:ApiKey"]);
    client.Timeout = TimeSpan.FromSeconds(15);
});

builder.Services.AddControllers();
var app = builder.Build();
app.MapControllers();
app.Run();

Named Client เหมาะสำหรับกรณีที่มีการเรียก Third-party หลายตัวหรือคอนฟิกหลายแบบ แต่ไม่ต้องการสร้าง Class Service แยกในแต่ละจุด

⚙️ Program.cs & การเรียกใช้งาน Named Client 
// 1. ลงทะเบียนใน Program.cs โดยตั้งชื่อระบุ Client
builder.Services.AddHttpClient("GitHubClient", client =>
{
    client.BaseAddress = new Uri("https://api.github.com/");
    client.DefaultRequestHeaders.Add("User-Agent", "DotNetLearningHub-App");
    client.DefaultRequestHeaders.Add("Accept", "application/vnd.github.v3+json");
});

// 2. เรียกใช้งานใน Controller ผ่าน IHttpClientFactory
[ApiController]
[Route("api/[controller]")]
public class RepoController : ControllerBase
{
    private readonly IHttpClientFactory _clientFactory;

    public RepoController(IHttpClientFactory clientFactory)
    {
        _clientFactory = clientFactory;
    }

    [HttpGet("{owner}/{repo}")]
    public async Task GetRepo(string owner, string repo)
    {
        // ดึง client ตามชื่อที่ตั้งไว้
        var client = _clientFactory.CreateClient("GitHubClient");
        
        var response = await client.GetAsync($"repos/{owner}/{repo}");
        if (!response.IsSuccessStatusCode)
            return StatusCode((int)response.StatusCode, "GitHub API Error");

        var json = await response.Content.ReadAsStringAsync();
        return Content(json, "application/json");
    }
}

Basic Pattern เหมาะสำหรับงานง่ายๆ เช่น Ad-hoc Request ที่ไม่มีการตั้งค่าล่วงหน้าใน Program.cs

📄 Services/AdHocService.cs 
public class QuickPingService
{
    private readonly IHttpClientFactory _factory;

    public QuickPingService(IHttpClientFactory factory)
    {
        _factory = factory;
    }

    public async Task IsSiteHealthyAsync(string url)
    {
        // สร้าง client แบบ default (ยังคงได้ประโยชน์จาก Handler Pool)
        var client = _factory.CreateClient();
        try
        {
            var response = await client.GetAsync(url);
            return response.IsSuccessStatusCode;
        }
        catch
        {
            return false;
        }
    }
}
03

JSON Serialization & Deserialization ด้วย System.Text.Json

การแปลงข้อมูล JSON ประสิทธิภาพสูง ไม่เปลือง Memory (Zero-allocation) และการจัดการ Naming Policy ระหว่าง C# PascalCase กับ API snake_case

System.Text.Json เป็น JSON Library ประจำตัวของ .NET ที่ถูกสร้างขึ้นมาเพื่อเน้นประสิทธิภาพความเร็วสูงกว่า Newtonsoft.Json (Json.NET) หลายเท่าตัว ด้วยการอ่านข้อมูลผ่าน Span<byte> และ UTF-8 โดยตรง

📌 กฎเหล็ก 3 ข้อที่ต้องรู้เมื่อแปลง JSON กับ Third-Party APIs

  • Property Naming: C# นิยมใช้ PascalCase (เช่น CustomerId) แต่ REST APIs สากลมักใช้ snake_case (เช่น customer_id) หรือ camelCase
  • Reusing JsonSerializerOptions: JsonSerializerOptions ควรประกาศเป็น static readonly เสมอ ห้าม new ทุกรอบ เพราะการสร้าง Options ใหม่จะสูญเสีย Reflection Cache ภายใน ทำให้โปรแกรมทำงานช้าลงอย่างเห็นได้ชัด
  • Async Streaming: ใช้ DeserializeAsync<T> ร่วมกับ Stream แทนที่จะอ่านเป็นสตริงทั้งหมดเข้า Memory ด้วย ReadAsStringAsync() เพื่อลดการบวมของ LOH (Large Object Heap)
📦 DTOs/UserProfileDto.cs (การตกแต่ง JSON Model ด้วย Attributes) 
using System.Text.Json.Serialization;

public class UserProfileDto
{
    // แมปชื่อจาก snake_case ใน JSON กลับมาเป็น PascalCase ใน C#
    [JsonPropertyName("user_id")]
    public required string Id { get; init; }

    [JsonPropertyName("full_name")]
    public string FullName { get; set; } = string.Empty;

    [JsonPropertyName("email_address")]
    public string? Email { get; set; }

    // ซ่อนฟิลด์นี้ ไม่ให้ถูก Serialize ส่งออกไปภายนอก
    [JsonIgnore]
    public string InternalSecretToken { get; set; } = string.Empty;

    // รองรับการแปลง Format วันที่แบบกำหนดเอง
    [JsonPropertyName("created_at")]
    public DateTime CreatedAt { get; set; }

    // กำหนดค่า Default หากฟิลด์ใน JSON ส่งมาเป็น null
    [JsonIgnore(Condition = JsonIgnoreCondition.WhenWritingNull)]
    [JsonPropertyName("phone_number")]
    public string? PhoneNumber { get; set; }
}
⚡ Services/JsonHelperExample.cs (การ Deserialization แบบ High-Performance Stream) 
using System.Text.Json;

public class ThirdPartyUserService
{
    private readonly HttpClient _client;
    
    // ตั้งค่า Options เพียงครั้งเดียวแบบ static เพื่อประหยัด CPU & Memory
    private static readonly JsonSerializerOptions _jsonOptions = new()
    {
        PropertyNameCaseInsensitive = true, // รองรับตัวพิมพ์เล็ก-ใหญ่
        PropertyNamingPolicy = JsonNamingPolicy.SnakeCaseLower, // แปลง snake_case อัตโนมัติ (.NET 8)
        WriteIndented = false
    };

    public ThirdPartyUserService(HttpClient client)
    {
        _client = client;
    }

    public async Task GetProfileStreamAsync(string userId, CancellationToken ct)
    {
        using var response = await _client.GetAsync($"/api/users/{userId}", HttpCompletionOption.ResponseHeadersRead, ct);
        response.EnsureSuccessStatusCode();

        // ดึง Stream โดยตรง ไม่ต้องอ่านเป็นสตริงทั้งหมดก่อน
        using var stream = await response.Content.ReadAsStreamAsync(ct);
        
        // Deserialize ตรงจาก Network Stream ทันที (Memory Footprint ต่ำมาก)
        return await JsonSerializer.DeserializeAsync(stream, _jsonOptions, ct);
    }
}
04

การจัดการ Resiliency & Fault Handling ด้วย Polly

รับมือกับปัญหาเครือข่ายล่มชั่วคราว (Transient Faults) ด้วยกลยุทธ์ Exponential Backoff + Jitter และ Circuit Breaker Pattern

ในการสื่อสารผ่านเครือข่ายระหว่าง Microservices หรือ Third-Party APIs มีกฎเหล็กข้อหนึ่งคือ "สิ่งใดที่สามารถผิดพลาดได้ มันจะผิดพลาดอย่างแน่นอน" (Network glitches, Server 503 Overloaded, Rate limit 429) เราต้องสร้างแอปที่มีความยืดหยุ่น (Resilient) ไม่พังตามระบบภายนอก

🔄

1. Retry Policy

เมื่อเกิดข้อผิดพลาดชั่วคราว (เช่น 503 Service Unavailable) ให้ลองยิงซ้ำตามจำนวนรอบที่กำหนด

⏳

2. Exponential Backoff

เว้นระยะห่างการลองใหม่แบบทวีคูณ (2s ➔ 4s ➔ 8s) พร้อมสุ่มเศษเสี้ยวเวลา (Jitter) ป้องกันคลื่น Request ถล่ม Server พร้อมกัน (Thundering Herd)

⚡

3. Circuit Breaker

เมื่อพังติดต่อกันเกินเกณฑ์ ให้ตัดวงจรทันที (Fail Fast) ไม่ส่ง Request ไปรบกวน Server ปลายทางที่กำลังล่ม

🛡️ Program.cs (การติดตั้ง Polly Resilience Pipeline ใน .NET 8) 
using Polly;
using Polly.Extensions.Http;

var builder = WebApplication.CreateBuilder(args);

// วิธีที่ 1: ติดตั้ง Policy ผ่าน Microsoft.Extensions.Http.Polly
builder.Services.AddHttpClient(client =>
{
    client.BaseAddress = new Uri("https://api.weather-service.com");
})
// 1. นโยบาย Retry: ทำซ้ำ 3 ครั้ง แบบ Exponential Backoff (2s, 4s, 8s) เฉพาะกรณี Transient Error (5xx, 408, Network Error)
.AddTransientHttpErrorPolicy(policyBuilder => 
    policyBuilder.WaitAndRetryAsync(
        retryCount: 3,
        sleepDurationProvider: retryAttempt => TimeSpan.FromSeconds(Math.Pow(2, retryAttempt))
    )
)
// 2. นโยบาย Circuit Breaker: ถ้าพังติดต่อกัน 5 ครั้ง ให้เปิดสวิตช์ตัดวงจรเป็นเวลา 30 วินาที
.AddTransientHttpErrorPolicy(policyBuilder =>
    policyBuilder.CircuitBreakerAsync(
        handledEventsAllowedBeforeBreaking: 5,
        durationOfBreak: TimeSpan.FromSeconds(30)
    )
);

// 🌟 วิธีที่ 2 (.NET 8 Standard Resilience Handler แนะนำใหม่ล่าสุด):

Webhooks Integration & การตรวจสอบ HMAC-SHA256 Signature

เทคนิคการสร้าง Webhook Receiver Endpoint ที่ปลอดภัย ป้องกัน Replay Attack และการปลอมแปลงข้อมูลจากบุคคลภายนอก

Webhook คือระบบ "Reverse API" แทนที่ระบบของเราจะต้องคอย Polling ไปถามเซิร์ฟเวอร์ภายนอกทุกๆ 5 นาทีว่า "ลูกค้าจ่ายเงินหรือยัง?" เมื่อเกิดเหตุการณ์ขึ้น เซิร์ฟเวอร์ต้นทาง (เช่น Stripe, Omise, GitHub, LINE Pay) จะเป็นผู้ยิง HTTP POST มาหา Endpoint ของเราโดยตรงทันทีแบบ Real-Time

1
ผู้ให้บริการสร้าง Digital Signature:
ก่อนส่งข้อมูล JSON ผู้ให้บริการจะนำ Raw JSON Body มารวมกับรหัสลับ (Shared Secret Key) แล้วคำนวณ Hash ด้วย HMAC-SHA256 จากนั้นแนบผลลัพธ์มาใน HTTP Header (เช่น X-Signature หรือ Stripe-Signature)
2
Receiver รับและคำนวณเปรียบเทียบ:
เมื่อ Endpoint ของเราได้รับ Request เราจะอ่าน Raw Body ที่แท้จริง แล้วนำมา Hash ด้วย Secret เดียวกัน หากผลลัพธ์ตรงกัน 100% จึงมั่นใจได้ว่าข้อมูลมาจากผู้ให้บริการตัวจริงและไม่มีการแก้ไขระหว่างทาง
🔒 Controllers/PaymentWebhookController.cs (Webhook Receiver ที่สมบูรณ์แบบ) 
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using Microsoft.AspNetCore.Mvc;

[ApiController]
[Route("api/webhooks")]
public class PaymentWebhookController : ControllerBase
{
    private readonly IConfiguration _config;
    private readonly ILogger _logger;

    public PaymentWebhookController(IConfiguration config, ILogger logger)
    {
        _config = config;
        _logger = logger;
    }

    [HttpPost("payment-callback")]
    public async Task HandlePaymentWebhook()
    {
        // 1. ดึง Signature Header ที่ส่งมาจาก Third-Party Gateway
        if (!Request.Headers.TryGetValue("X-Signature", out var signatureHeader))
        {
            _logger.LogWarning("ปฏิเสธคำขอ: ไม่พบ X-Signature Header");
            return Unauthorized("Missing signature header");
        }

        // 2. อ่าน Raw Body แบบดิบๆ (ต้องเป็น Bytes เดิมเป๊ะๆ ห้ามแปลงเป็น Object ก่อน)
        using var reader = new StreamReader(Request.Body, Encoding.UTF8);
        var rawPayload = await reader.ReadToEndAsync();

        // 3. ตรวจสอบความถูกต้องของ Signature ด้วย HMAC-SHA256
        var secretKey = _config["PaymentGateway:WebhookSecret"] ?? "super_secret_webhook_key";
        if (!VerifyHmacSignature(rawPayload, signatureHeader!, secretKey))
        {
            _logger.LogWarning("คำขอถูกแฮกหรือปลอมแปลง: Signature ไม่ตรงกัน!");
            return Unauthorized("Invalid signature");
        }

        // 4. เมื่อผ่านการยืนยันตัวตนแล้ว จึงนำ Payload ไปประมวลผลทางธุรกิจ
        var eventData = JsonSerializer.Deserialize(rawPayload);
        var eventType = eventData.GetProperty("event_type").GetString();
        var orderId = eventData.GetProperty("data").GetProperty("order_id").GetString();

        _logger.LogInformation("รับ Webhook สำเร็จ! Event: {EventType}, OrderId: {OrderId}", eventType, orderId);

        // ตอบกลับ 200 OK ทันที เพื่อบอก Gateway ว่ารับของเรียบร้อย
        return Ok(new { received = true });
    }

    private static bool VerifyHmacSignature(string payload, string expectedSignature, string secret)
    {
        using var hmac = new HMACSHA256(Encoding.UTF8.GetBytes(secret));
        var hashBytes = hmac.ComputeHash(Encoding.UTF8.GetBytes(payload));
        var computedSignature = Convert.ToHexString(hashBytes).ToLowerInvariant();

        // ใช้ CryptographicOperations.FixedTimeEquals เพื่อป้องกัน Timing Attack
        return CryptographicOperations.FixedTimeEquals(
            Encoding.UTF8.GetBytes(computedSignature),
            Encoding.UTF8.GetBytes(expectedSignature.ToLowerInvariant())
        );
    }
}
