ออกแบบและพัฒนา RESTful API ด้วย ASP.NET Core

มาตรฐานการออกแบบ REST API, การเลือกใช้ HTTP Verbs, การส่ง HTTP Status Codes ที่ถูกต้อง, การใช้ DTOs และ Model Binding

⭐ ระดับ: กลาง ⏱️ เวลา: ~50 นาที ⚡ รางวัล: +100 XP 🏷️ REST / HTTP Status Codes / DTO
🍽️ เปรียบเทียบให้เห็นภาพ: ร้านอาหารตามสั่งมาตรฐานสากล
💡 Analogy

• API Endpoint URL: เหมือน หมวดรายการในเมนู เช่น /api/orders (ใช้คำนามพหูพจน์)
• HTTP Verb: เหมือน การกระทำของลูกค้า เช่น GET (ขอดูเมนู), POST (สั่งทำจานใหม่), PUT (เปลี่ยนรายการทั้งจาน), DELETE (ขอยกเลิก)
• Controller: เหมือน พนักงานเสิร์ฟและพ่อครัว ที่รับออเดอร์ไปปรุงอาหาร
• HTTP Status Code: เหมือน คำตอบของบริกร เช่น 200 OK (อาหารมาแล้วครับ), 201 Created (อาหารจานใหม่ทำเสร็จแล้ว), 404 Not Found (ไม่มีเมนูนี้), 400 Bad Request (สั่งอาหารผิดเงื่อนไข)

01

HTTP Verbs & Idempotency (ความคงที่ของผลลัพธ์)

ความคงที่ (Idempotent) หมายถึง การยิงคำขอนั้นซ้ำหลายครั้ง ผลลัพธ์บนเซิร์ฟเวอร์ยังคงเหมือนเดิม:

Verb หน้าที่ Idempotent? Safe? (ไม่แก้ข้อมูล) Success Status
GET ดึงข้อมูล (Read) ✅ ใช่ ✅ ปลอดภัย 200 OK
POST สร้างข้อมูลใหม่ (Create) ❌ ไม่ใช่ (ยิงซ้ำสร้างเพิ่ม) ❌ มีการแก้ข้อมูล 201 Created
PUT แก้ไข/แทนที่ทั้ง Record (Replace) ✅ ใช่ (แทนที่ด้วยค่าเดิม) ❌ มีการแก้ข้อมูล 200 OK หรือ 204 No Content
PATCH แก้ไขเฉพาะบางฟิลด์ (Partial Update) ❌ อาจไม่ใช่ ❌ มีการแก้ข้อมูล 200 OK หรือ 204 No Content
DELETE ลบข้อมูล (Delete) ✅ ใช่ (ลบแล้วก็หายไป) ❌ มีการแก้ข้อมูล 200 OK หรือ 204 No Content
02

HTTP Status Codes ที่ Junior .NET ต้องรู้จริง

🟢 2xx Success

  • 200 OK: สำเร็จ ส่งข้อมูลกลับ
  • 201 Created: สร้าง Resource ใหม่สำเร็จ พร้อม Header Location
  • 204 No Content: สำเร็จ แต่ไม่มี Body ตอบกลับ (เช่น หลัง DELETE)

🟡 4xx Client Errors

  • 400 Bad Request: ข้อมูล Input ไม่ถูกต้อง (Validation Error)
  • 401 Unauthorized: ยังไม่ได้ Login (No Token / Invalid Token)
  • 403 Forbidden: Login แล้วแต่ไม่มีสิทธิ์เข้าถึง Role นี้
  • 404 Not Found: ไม่พบข้อมูลในระบบ
  • 409 Conflict: ข้อมูลซ้ำกัน (เช่น อีเมลนี้มีในระบบแล้ว)

🔴 5xx Server Errors

  • 500 Internal Server Error: เกิด Unhandled Exception ในเซิร์ฟเวอร์
  • 502 Bad Gateway: เซิร์ฟเวอร์ต้นทางได้รับคำตอบที่ผิดพลาดจากบริการอื่น
  • 503 Service Unavailable: ระบบปิดปรับปรุงชั่วคราว
03

ตัวอย่าง Controller & การใช้ DTOs ในงานจริง

📄 ProductsController.cs
[ApiController]
[Route("api/v1/[controller]")]
public class ProductsController : ControllerBase
{
    private readonly IProductService _service;

    public ProductsController(IProductService service)
    {
        _service = service;
    }

    // GET: api/v1/products?category=software
    [HttpGet]
    public async Task<ActionResult<IEnumerable<ProductResponseDto>>> GetProducts([FromQuery] string? category)
    {
        var products = await _service.GetProductsAsync(category);
        return Ok(products); // 200 OK
    }

    // GET: api/v1/products/5
    [HttpGet("{id:int}")]
    public async Task<ActionResult<ProductResponseDto>> GetById(int id)
    {
        var product = await _service.GetByIdAsync(id);
        if (product == null)
        {
            return NotFound(new { message = $"ไม่พบสินค้า ID {id}" }); // 404 Not Found
        }
        return Ok(product); // 200 OK
    }

    // POST: api/v1/products
    [HttpPost]
    public async Task<ActionResult<ProductResponseDto>> Create([FromBody] CreateProductRequestDto request)
    {
        if (!ModelState.IsValid)
        {
            return BadRequest(ModelState); // 400 Bad Request
        }

        var created = await _service.CreateProductAsync(request);
        // 201 Created พร้อม Header Location ชี้ไปที่ GetById
        return CreatedAtAction(nameof(GetById), new { id = created.Id }, created);
    }

    // DELETE: api/v1/products/5
    [HttpDelete("{id:int}")]
    public async Task<IActionResult> Delete(int id)
    {
        var success = await _service.DeleteProductAsync(id);
        if (!success) return NotFound();
        return NoContent(); // 204 No Content
    }
}
