01

คลาสและออบเจกต์ (Class & Object)

รากฐานที่สำคัญที่สุดของภาษา C# — เปลี่ยนแบบแปลนความคิดให้กลายเป็นวัตถุจริงในหน่วยความจำ

💡 เปรียบเทียบให้เห็นภาพ (Analogy)
📐 แบบพิมพ์เขียว (Class) vs บ้านที่สร้างเสร็จแล้ว (Object)

• Class (คลาส) คือ "แบบพิมพ์เขียวของบ้าน" (Blueprint) — บอกว่าบ้านนี้ต้องมีกี่ห้องนอน มีประตู มีหน้าต่าง มีฟังก์ชันเปิดไฟ แต่ตัวพิมพ์เขียวเองไม่ใช่บ้านจริง ไม่เปลืองที่ดิน (ยังไม่จองหน่วยความจำ RAM จริง)
• Object (ออบเจกต์) คือ "บ้านจริงที่สร้างขึ้นมา" (Instance) — สร้างจากแบบพิมพ์เขียวโดยใช้คำสั่ง new มีเลขที่บ้านเป็นของตัวเอง (Memory Address) และคุณสามารถสร้างบ้านได้กี่หลังก็ได้จากแบบพิมพ์เขียวใบเดียว!

📐 Class: Product
• string Name
• decimal Price
• void DisplayDetails()
* นิยามโครงสร้างและพฤติกรรม
new ➔
📦 Objects จริงใน Heap RAM
1. laptop (Name="MacBook", Price=45,000)
2. mouse (Name="Logitech", Price=1,290)
* แต่ละตัวมีข้อมูลแยกกันอิสระ
📄 Product.cs 
using System;

namespace DotNetLearningHub.OOP
{
    // 1. ประกาศ Class (แบบพิมพ์เขียวสินค้า)
    public class Product
    {
        // Properties: คุณลักษณะของสินค้า
        public string Name { get; set; }
        public decimal Price { get; set; }
        public int Stock { get; set; }

        // Constructor: เมธอดพิเศษที่ทำงานอัตโนมัติเมื่อสร้าง Object ด้วย new
        public Product(string name, decimal price, int stock)
        {
            Name = name;
            Price = price;
            Stock = stock;
        }

        // Method: พฤติกรรมหรือความสามารถของสินค้า
        public void DisplayDetails()
        {
            Console.WriteLine($"📦 สินค้า: {Name,-15} | ราคา: {Price,8:N2} บาท | คงเหลือ: {Stock,3} ชิ้น");
        }

        public void ApplyDiscount(decimal percentage)
        {
            decimal discountAmount = Price * (percentage / 100);
            Price -= discountAmount;
            Console.WriteLine($"🏷️ ลดราคา {percentage}% ให้สินค้า {Name}! ราคาใหม่คือ: {Price:N2} บาท");
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("=== การสร้างและใช้งาน Object ใน C# ===\n");

            // 2. สร้าง Object ตัวแรกผ่าน Constructor
            Product laptop = new Product("Gaming Laptop", 35900.00m, 5);

            // 3. สร้าง Object ตัวที่สอง
            Product mouse = new Product("Wireless Mouse", 1290.00m, 20);

            // 4. เรียกใช้ Method ของแต่ละ Object
            laptop.DisplayDetails();
            mouse.DisplayDetails();

            Console.WriteLine();
            // จัดโปรโมชันลดราคา 10% ให้เฉพาะ Laptop
            laptop.ApplyDiscount(10);

            Console.WriteLine("\n--- ข้อมูลหลังจัดโปรโมชัน ---");
            laptop.DisplayDetails();
            mouse.DisplayDetails();
        }
    }
}

02

การห่อหุ้มข้อมูลและระดับการเข้าถึง (Encapsulation)

ปกป้องข้อมูลสำคัญไม่ให้ถูกแก้ไขโดยพลการ พร้อมตรวจสอบความถูกต้อง (Validation) ก่อนบันทึก

🔒 ห้องส่วนตัว vs ห้องรับแขก (Encapsulation Analogy)
🏠 เปรียบเทียบพื้นที่ในบ้านกับการเข้าถึงข้อมูล

• public (ห้องรับแขก) — แขกทุกคนที่มาเยือนสามารถเข้ามานั่งพักผ่อนได้อิสระ โค้ดจากภายนอกคลาสเรียกใช้งานได้โดยตรง
• private (ห้องนอนส่วนตัว / ตู้เซฟ) — มีเฉพาะเจ้าของห้องเท่านั้นที่มีกุญแจไขเข้าไปได้ โค้ดภายนอกมองไม่เห็นและแตะต้องไม่ได้
• protected (สมบัติที่ส่งต่อให้ลูกหลาน) — คนในครอบครัวและลูกหลานที่สืบทอดสายเลือดเท่านั้นที่เข้าถึงได้ (คลาสแม่และคลาสลูก)
• internal (ประตูรั้วหมู่บ้าน) — ใครก็ตามที่อยู่ในหมู่บ้านเดียวกัน (Assembly / Project เดียวกัน) สามารถเข้าถึงได้

public

เข้าถึงได้จากทุกที่ ทั้งภายในและภายนอกโปรเจกต์

private

เข้าถึงได้เฉพาะ ภายในคลาสเดียวกัน เท่านั้น (ค่าเริ่มต้นของ Field)

protected

เข้าถึงได้ภายในคลาสตัวเอง และ คลาสลูกที่สืบทอดไป

internal

เข้าถึงได้เฉพาะโค้ดที่อยู่ใน Assembly/Project เดียวกัน

📄 BankAccount.cs 
using System;

namespace DotNetLearningHub.OOP
{
    public class BankAccount
    {
        // 1. Backing Field เก็บเป็น private เพื่อป้องกันไม่ให้ภายนอกแก้ตัวเลขโดยตรง
        private decimal _balance;
        private string _accountNumber;

        // 2. Public Property ควบคุมการอ่าน (get) และเขียน (set) พร้อม Validation
        public string AccountNumber 
        { 
            get => _accountNumber;
            // init-only: กำหนดได้เฉพาะตอนเปิดบัญชีเท่านั้น หลังจากนั้นห้ามแก้ไข!
            init => _accountNumber = value; 
        }

        public string AccountHolder { get; set; }

        // Property ยอดเงินคงเหลือ: ให้อ่านได้อย่างเดียว (Read-Only จากภายนอก)
        public decimal Balance 
        { 
            get { return _balance; } 
            private set 
            { 
                if (value < 0)
                {
                    throw new ArgumentException("ยอดเงินคงเหลือในบัญชีต้องไม่ติดลบ!");
                }
                _balance = value; 
            }
        }

        // Constructor
        public BankAccount(string accNo, string holder, decimal initialDeposit)
        {
            if (initialDeposit < 500)
            {
                throw new ArgumentException("เปิดบัญชีขั้นต่ำต้องฝาก 500 บาทขึ้นไป");
            }

            AccountNumber = accNo;
            AccountHolder = holder;
            _balance = initialDeposit;
        }

        // Method ฝากเงิน: มีการตรวจสอบค่าก่อนทำรายการ
        public void Deposit(decimal amount)
        {
            if (amount <= 0)
            {
                Console.WriteLine("⚠️ ยอดเงินฝากต้องมากกว่า 0 บาท!");
                return;
            }

            _balance += amount;
            Console.WriteLine($"✅ ฝากเงินสำเร็จ: +{amount:N2} บาท | ยอดคงเหลือปัจจุบัน: {_balance:N2} บาท");
        }

        // Method ถอนเงิน: ตรวจสอบว่ายอดเงินในบัญชีมีเพียงพอหรือไม่
        public bool Withdraw(decimal amount)
        {
            if (amount <= 0)
            {
                Console.WriteLine("⚠️ ยอดเงินถอนต้องมากกว่า 0 บาท!");
                return false;
            }

            if (amount > _balance)
            {
                Console.WriteLine($"❌ ถอนเงินไม่สำเร็จ: ยอดเงินไม่พอ (ต้องการถอน {amount:N2} แต่มี {_balance:N2})");
                return false;
            }

            _balance -= amount;
            Console.WriteLine($"💸 ถอนเงินสำเร็จ: -{amount:N2} บาท | ยอดคงเหลือปัจจุบัน: {_balance:N2} บาท");
            return true;
        }
    }

    class Program
    {
        static void Main()
        {
            Console.WriteLine("=== ระบบบัญชีธนาคาร (Encapsulation Demo) ===\n");

            var account = new BankAccount("ACC-90210", "สมชาย สายโค้ด", 1000m);
            Console.WriteLine($"เจ้าของบัญชี: {account.AccountHolder} | เลขที่: {account.AccountNumber}");
            Console.WriteLine($"ยอดเงินเริ่มต้น: {account.Balance:N2} บาท\n");

            // ทดลองทำรายการฝาก-ถอนที่ถูกต้อง
            account.Deposit(2500m);
            account.Withdraw(1200m);

            // ทดลองถอนเงินเกินบัญชี (ระบบจะดักจับไว้ได้ ป้องกันเงินติดลบ)
            account.Withdraw(50000m);

            // account._balance = 999999; // ❌ คอมไพล์ไม่ผ่านทันที! เพราะ _balance เป็น private
        }
    }
}
▶ Output: === ระบบบัญชีธนาคาร (Encapsulation Demo) === เจ้าของบัญชี: สมชาย สายโค้ด | เลขที่: ACC-90210 ยอดเงินเริ่มต้น: 1,000.00 บาท ✅ ฝากเงินสำเร็จ: +2,500.00 บาท | ยอดคงเหลือปัจจุบัน: 3,500.00 บาท 💸 ถอนเงินสำเร็จ: -1,200.00 บาท | ยอดคงเหลือปัจจุบัน: 2,300.00 บาท ❌ ถอนเงินไม่สำเร็จ: ยอดเงินไม่พอ (ต้องการถอน 50,000.00 แต่มี 2,300.00)
⚠️ กฎเหล็ก: ทำไมไม่ควรตั้ง Field เป็น Public โดยตรง?

หากเราเปิด public decimal Balance; โค้ดภายนอกจะสามารถแอบเขียน acc.Balance = -999999; ได้ทันทีโดยไม่มีการตรวจเช็ก ส่งผลให้เกิด Bug ร้ายแรงในระบบบัญชี! การใช้ Encapsulation จึงเป็นการสร้างด่านตรวจความปลอดภัยให้กับข้อมูลเสมอ

03

การสืบทอดคุณสมบัติ (Inheritance)

ลดความซ้ำซ้อนของโค้ด (DRY) โดยให้คลาสลูกสืบทอดและต่อยอดจากคลาสแม่

🧬 พ่อแม่ถ่ายทอดยีนให้ลูก (Inheritance Analogy)
👨‍👩‍👧 การส่งต่อพันธุกรรมและความสามารถ

• Base Class (คลาสแม่ / คลาสหลัก) เช่น Animal — มีคุณสมบัติพื้นฐานที่สัตว์ทุกตัวต้องมีร่วมกัน เช่น มีชื่อ (Name), มีอายุ (Age), หายใจได้, กินอาหารได้ (Eat())
• Derived Class (คลาสลูก) เช่น Dog และ Cat — ได้รับคุณสมบัติทั้งหมดจากแม่โดยไม่ต้องเขียนซ้ำ และสามารถเพิ่มความสามารถเฉพาะตัว เช่น สุนัขมีเมธอด FetchBall() คาบลูกบอลได้ ส่วนแมวมี ClimbTree() ปีนต้นไม้ได้
• คีย์เวิร์ด : base(...) คือการส่งค่าขึ้นไปให้ Constructor ของคลาสแม่ทำงานก่อน

📄 Animals.cs 
using System;

namespace DotNetLearningHub.OOP
{
    // 1. Base Class (คลาสแม่)
    public class Animal
    {
        public string Name { get; set; }
        public int Age { get; set; }

        public Animal(string name, int age)
        {
            Name = name;
            Age = age;
        }

        public void Eat()
        {
            Console.WriteLine($"🍽️ {Name} กำลังกินอาหารอย่างเอร็ดอร่อย");
        }

        // คีย์เวิร์ด virtual อนุญาตให้คลาสลูกเขียนทับ (override) พฤติกรรมได้
        public virtual void MakeSound()
        {
            Console.WriteLine($"🔊 {Name} กำลังส่งเสียงทั่วไป");
        }
    }

    // 2. Derived Class 1 (คลาสลูก Dog สืบทอดจาก Animal ด้วยเครื่องหมาย :)
    public class Dog : Animal
    {
        public string Breed { get; set; }

        // : base(name, age) ส่งพารามิเตอร์ขึ้นไปให้ Animal Constructor จัดการ
        public Dog(string name, int age, string breed) : base(name, age)
        {
            Breed = breed;
        }

        // คีย์เวิร์ด override เขียนทับการส่งเสียงให้เป็นเสียงเห่า
        public override void MakeSound()
        {
            Console.WriteLine($"🐕 {Name} (พันธุ์ {Breed}) เห่า: โฮ่งๆ! บ๊อกๆ!");
        }

        // เมธอดเฉพาะตัวของสุนัข
        public void FetchStick()
        {
            Console.WriteLine($"🎾 {Name} วิ่งไปคาบกิ่งไม้กลับมาอย่างรวดเร็ว!");
        }
    }

    // 3. Derived Class 2 (คลาสลูก Cat สืบทอดจาก Animal)
    public class Cat : Animal
    {
        public bool IsLazy { get; set; }

        public Cat(string name, int age, bool isLazy) : base(name, age)
        {
            IsLazy = isLazy;
        }

        public override void MakeSound()
        {
            Console.WriteLine($"🐈 {Name} ร้อง: เหมียว~ เหมียว~ ฟี้...");
        }

        public void ScratchSofa()
        {
            Console.WriteLine($"🛋️ {Name} กำลังลับเล็บกับโซฟาจนเป็นรอย!");
        }
    }

    class Program
    {
        static void Main()
        {
            Console.WriteLine("=== การสืบทอดคุณสมบัติ (Inheritance Demo) ===\n");

            Dog lucky = new Dog("ลัคกี้", 3, "โกลเด้น รีทรีฟเวอร์");
            lucky.Eat();         // เมธอดที่สืบทอดมาจาก Animal
            lucky.MakeSound();   // เมธอดที่ override ใหม่
            lucky.FetchStick();  // เมธอดเฉพาะของ Dog

            Console.WriteLine();

            Cat mimi = new Cat("มี่มี่", 2, true);
            mimi.Eat();          // เมธอดที่สืบทอดมาจาก Animal
            mimi.MakeSound();    // เมธอดที่ override ใหม่
            mimi.ScratchSofa();  // เมธอดเฉพาะของ Cat
        }
    }
}

💡 รู้หรือไม่: Single Inheritance ใน C#

ภาษา C# ยอมให้คลาสหนึ่งสืบทอดจาก คลาสแม่ได้เพียงคลาสเดียวเท่านั้น (Single Class Inheritance) เพื่อป้องกันความสับสนเรื่อง Diamond Problem แต่สามารถ Implement หลาย Interface พร้อมกันได้ไม่จำกัด!

04

การมีได้หลายรูปแบบ (Polymorphism)

เรียกใช้คำสั่งเดียวกันผ่านตัวแปรประเภทแม่ แต่ผลลัพธ์ปรับเปลี่ยนตามประเภทของออบเจกต์ลูกจริง

🎭 หน้ากากเดียวกัน แต่พฤติกรรมต่างกัน (Polymorphism Analogy)
🎮 ปุ่มกด Action เดียวกันบนจอยเกม

ลองนึกถึงปุ่ม [X - Attack] บนจอยเกมคอนโซล:
• ถ้าตัวละครปัจจุบันเป็น อัศวิน (Knight) ➔ ตัวละครจะ กวัดแกว่งดาบ
• ถ้าตัวละครเป็น นักธนู (Archer) ➔ ตัวละครจะ ยิงลูกธนู
• ถ้าตัวละครเป็น จอมเวท (Mage) ➔ ตัวละครจะ ร่ายคาถาลูกไฟ
ระบบเกมไม่ต้องเขียน if-else แยก 100 ชนิดตัวละคร แค่สั่ง hero.Attack() ตัวละครจะตอบสนองด้วยท่าทางเฉพาะตัวของมันเอง นี่คือหัวใจของ Dynamic Polymorphism

📄 PolymorphismDemo.cs 
using System;
using System.Collections.Generic;

namespace DotNetLearningHub.OOP
{
    // คลาสลูกตัวที่ 3
    public class Bird : Animal
    {
        public Bird(string name, int age) : base(name, age) { }

        public override void MakeSound()
        {
            Console.WriteLine($"🐦 {Name} ส่งเสียงร้อง: จิ๊บๆ! จิ๊บๆ! 🎶");
        }
    }

    class Program
    {
        static void Main()
        {
            Console.WriteLine("=== การทำงานของ Runtime Polymorphism ===\n");

            // เก็บออบเจกต์ต่างชนิดกัน (Dog, Cat, Bird) ไว้ใน List เดียวกัน
            List farmZoo = new List()
            {
                new Dog("เจ้าตูบ", 4, "ไซบีเรียน"),
                new Cat("เจ้าส้ม", 2, false),
                new Bird("นกแก้วโนริ", 1),
                new Dog("โกโก้", 5, "พุดเดิ้ล")
            };

            Console.WriteLine("📢 เรียกคำสั่ง animal.MakeSound() วนลูปทุกตัว:");
            Console.WriteLine("---------------------------------------------");

            // วนลูปสั่งงานด้วยคำสั่งเดียวกันทั้งหมด!
            foreach (Animal animal in farmZoo)
            {
                // .NET Runtime จะตัดสินใจเรียก MakeSound() ของตัวจริง (Dog/Cat/Bird) ณ ตอนรัน
                animal.MakeSound();
            }
        }
    }
}
▶ Output: === การทำงานของ Runtime Polymorphism === 📢 เรียกคำสั่ง animal.MakeSound() วนลูปทุกตัว: --------------------------------------------- 🐕 เจ้าตูบ (พันธุ์ ไซบีเรียน) เห่า: โฮ่งๆ! บ๊อกๆ! 🐈 เจ้าส้ม ร้อง: เหมียว~ เหมียว~ ฟี้... 🐦 นกแก้วโนริ ส่งเสียงร้อง: จิ๊บๆ! จิ๊บๆ! 🎶 🐕 โกโก้ (พันธุ์ พุดเดิ้ล) เห่า: โฮ่งๆ! บ๊อกๆ!
💡 ความแตกต่าง: Overriding vs Overloading

• Method Overriding (Dynamic Polymorphism): คลาสลูกเขียนทับเมธอดของคลาสแม่ (ชื่อเหมือน, พารามิเตอร์เหมือนกันทุกอย่าง) ทำงานตอน Run-time
• Method Overloading (Static Polymorphism): เขียนเมธอดชื่อเดียวกันภายในคลาสเดียวกัน แต่เปลี่ยนจำนวนหรือชนิดข้อมูลของ Parameter เช่น Add(int a, int b) กับ Add(double a, double b) ตัดสินใจตอน Compile-time

05

อินเทอร์เฟซและสัญญาการทำงาน (Interface)

กำหนดสัญญา (Contract) บังคับว่าคลาสที่นำไปใช้ต้องมีเมธอดอะไรบ้าง รองรับการต่อแบบแยกชิ้นส่วน (Loose Coupling)

🔌 ปลั๊กไฟและเต้ารับสากล (Interface Analogy)
⚡ สัญญารูปทรงของเต้ารับไฟฟ้า

• Interface เหมือน "มาตรฐานเต้าเสียบปลั๊กไฟ 3 ขา" — ระบุแค่ว่าใครจะเสียบต้องมี 3 ขาและรับไฟ 220V แต่ไม่สนใจว่าอุปกรณ์ที่นำมาเสียบจะเป็น พัดลม, ตู้เย็น, ไมโครเวฟ หรือคอมพิวเตอร์
• Convention ใน C#: ชื่อ Interface จะขึ้นต้นด้วยตัวอักษร I เสมอ เช่น IPayable, ILogger, IDisposable, IRepository
• Interface มีแต่หัวเมธอด (Method Signature) ไม่มีการเขียนโค้ดการทำงาน (Implementation) จนกระทั่งคลาสที่เอาไปใช้นำไปเติมเนื้อหาจริง

📄 PaymentService.cs 
using System;

namespace DotNetLearningHub.OOP
{
    // 1. ประกาศ Interface สัญญาสำหรับการชำระเงิน
    public interface IPayable
    {
        string PaymentProvider { get; }
        bool ProcessPayment(decimal amount);
    }

    // 2. ประกาศ Interface เพิ่มเติมสำหรับการออกใบเสร็จ
    public interface IReceiptGenerator
    {
        void PrintReceipt(string transactionId, decimal amount);
    }

    // 3. คลาสชำระผ่านบัตรเครดิต (Implement ทั้ง 2 Interface)
    public class CreditCardPayment : IPayable, IReceiptGenerator
    {
        public string PaymentProvider => "VISA / Mastercard Gateway";
        public string CardNumber { get; set; }

        public CreditCardPayment(string cardNumber)
        {
            CardNumber = cardNumber;
        }

        public bool ProcessPayment(decimal amount)
        {
            string maskedCard = CardNumber.Substring(CardNumber.Length - 4);
            Console.WriteLine($"💳 กำลังตัดบัตรเครดิตหมายเลข ****-****-****-{maskedCard} ยอด: {amount:N2} บาท");
            Console.WriteLine($"   ➔ ผ่านระบบ {PaymentProvider} สำเร็จ!");
            return true;
        }

        public void PrintReceipt(string transactionId, decimal amount)
        {
            Console.WriteLine($"   🧾 [E-Receipt #{transactionId}]: ยอดชำระ {amount:N2} บาท (สถานะ: ชำระแล้ว)\n");
        }
    }

    // 4. คลาสชำระผ่านพร้อมเพย์ QR Code
    public class PromptPayPayment : IPayable, IReceiptGenerator
    {
        public string PaymentProvider => "Thai QR Payment (PromptPay)";
        public string PhoneNumber { get; set; }

        public PromptPayPayment(string phoneNumber)
        {
            PhoneNumber = phoneNumber;
        }

        public bool ProcessPayment(decimal amount)
        {
            Console.WriteLine($"📱 สร้าง QR Code พร้อมเพย์เบอร์ {PhoneNumber} ยอด: {amount:N2} บาท");
            Console.WriteLine($"   ➔ ลูกค้าสแกนและโอนเงินเรียบร้อยผ่าน {PaymentProvider}");
            return true;
        }

        public void PrintReceipt(string transactionId, decimal amount)
        {
            Console.WriteLine($"   🧾 [PromptPay Slip #{transactionId}]: ยอดเงินเข้าบัญชี {amount:N2} บาท เรียบร้อย\n");
        }
    }

    // 5. คลาส Checkout ในระบบ E-Commerce ที่เรียกใช้งานผ่าน Interface (Loose Coupling)
    public class CheckoutService
    {
        public void CompleteOrder(IPayable paymentMethod, decimal totalAmount)
        {
            Console.WriteLine($"🛒 เริ่มกระบวนการชำระเงินยอดรวม: {totalAmount:N2} บาท");
            bool isSuccess = paymentMethod.ProcessPayment(totalAmount);

            if (isSuccess && paymentMethod is IReceiptGenerator receiptService)
            {
                string txId = "TXN-" + Guid.NewGuid().ToString().Substring(0, 8).ToUpper();
                receiptService.PrintReceipt(txId, totalAmount);
            }
        }
    }

    class Program
    {
        static void Main()
        {
            Console.WriteLine("=== ระบบชำระเงินด้วย Interface (Polymorphic Payments) ===\n");

            var checkout = new CheckoutService();

            // ชำระด้วยบัตรเครดิต
            IPayable card = new CreditCardPayment("4532112233448899");
            checkout.CompleteOrder(card, 3500m);

            // ชำระด้วย PromptPay
            IPayable qr = new PromptPayPayment("081-234-5678");
            checkout.CompleteOrder(qr, 790m);
        }
    }
}
▶ Output: === ระบบชำระเงินด้วย Interface (Polymorphic Payments) === 🛒 เริ่มกระบวนการชำระเงินยอดรวม: 3,500.00 บาท 💳 กำลังตัดบัตรเครดิตหมายเลข ****-****-****-8899 ยอด: 3,500.00 บาท ➔ ผ่านระบบ VISA / Mastercard Gateway สำเร็จ! 🧾 [E-Receipt #TXN-7A8B9C0D]: ยอดชำระ 3,500.00 บาท (สถานะ: ชำระแล้ว) 🛒 เริ่มกระบวนการชำระเงินยอดรวม: 790.00 บาท 📱 สร้าง QR Code พร้อมเพย์เบอร์ 081-234-5678 ยอด: 790.00 บาท ➔ ลูกค้าสแกนและโอนเงินเรียบร้อยผ่าน Thai QR Payment (PromptPay) 🧾 [PromptPay Slip #TXN-1F2E3D4C]: ยอดเงินเข้าบัญชี 790.00 บาท เรียบร้อย

📊 ตารางเปรียบเทียบ: Interface vs Abstract Class

คุณสมบัติ 🔌 Interface 🏛️ Abstract Class
เป้าหมายหลัก กำหนด "สัญญา (Can-Do)" ว่าทำอะไรได้บ้าง สร้าง "โครงสร้างแกนกลาง (Is-A)" ของตระกูลคลาส
การสืบทอด (Inheritance) หลายตัวได้ (Multiple) คลาสหนึ่ง implement ได้หลาย interface ได้เพียงคลาสเดียว (Single) ต่อหนึ่งคลาสลูก
มี Constructor หรือไม่? ❌ ไม่มี Constructor ✅ มี Constructor เพื่อเซ็ตค่าเริ่มต้นให้คลาสแม่ได้
State & Fields ❌ ไม่มี Field เก็บสถานะตัวแปรทั่วไป (มีได้เฉพาะ Property/Method Signature) ✅ สามารถมี Field, Protected Variable และ State ภายในได้
โค้ดการทำงานจริง เดิมไม่มี (ใน C# 8+ มี Default Interface Method ได้) ✅ ผสมผสานได้ทั้ง abstract method และ concrete method ทั่วไป
06

คลาสนามธรรม (Abstract Class)

คลาสแม่แบบที่ไม่สามารถสร้างออบเจกต์ตรงๆ ได้ ออกแบบมาเพื่อเป็นฐานให้คลาสอื่นนำไปสืบทอดและเติมเต็ม

📐 รูปทรงเรขาคณิตทั่วไป (Shape Analogy)
🔺 แนวคิดรูปทรงกับความเป็นจริง

หากมีคนสั่งคุณว่า "ช่วยวาดรูปทรงเรขาคณิต (Shape) ให้ดูหน่อยสิ" คุณจะถามกลับทันทีว่า "จะให้วาดวงกลม สามเหลี่ยม หรือสี่เหลี่ยม?"
เพราะคำว่า "รูปทรง" เป็นเพียงนามธรรม (Abstract) ไม่มีสูตรคำนวณพื้นที่ตายตัวจนกว่าจะระบุว่าเป็นรูปทรงอะไรกันแน่
ดังนั้นเราจึงสร้าง abstract class Shape ไว้เป็นแม่แบบ และให้ Circle และ Rectangle นำไป Implement สูตรคำนวณ CalculateArea() ของตนเอง

📄 Shapes.cs 
using System;

namespace DotNetLearningHub.OOP
{
    // 1. ประกาศ Abstract Class (ห้ามใช้ new Shape() ตรงๆ)
    public abstract class Shape
    {
        public string Color { get; set; }

        protected Shape(string color)
        {
            Color = color;
        }

        // Abstract Method: ไม่มีเนื้อโค้ด บังคับให้คลาสลูกต้อง override เสมอ!
        public abstract double CalculateArea();
        public abstract double CalculatePerimeter();

        // Concrete Method: มีโค้ดพร้อมใช้งาน คลาสลูกเรียกใช้ได้ทันที
        public void DisplayInfo()
        {
            Console.WriteLine($"🎨 รูปทรงสี: {Color}");
            Console.WriteLine($"   📐 พื้นที่: {CalculateArea():N2} ตร.หน่วย");
            Console.WriteLine($"   📏 ความยาวรอบรูป: {CalculatePerimeter():N2} หน่วย\n");
        }
    }

    // 2. Concrete Class 1: วงกลม
    public class Circle : Shape
    {
        public double Radius { get; set; }

        public Circle(string color, double radius) : base(color)
        {
            Radius = radius;
        }

        public override double CalculateArea()
        {
            return Math.PI * Radius * Radius;
        }

        public override double CalculatePerimeter()
        {
            return 2 * Math.PI * Radius;
        }
    }

    // 3. Concrete Class 2: สี่เหลี่ยมผืนผ้า
    public class Rectangle : Shape
    {
        public double Width { get; set; }
        public double Height { get; set; }

        public Rectangle(string color, double width, double height) : base(color)
        {
            Width = width;
            Height = height;
        }

        public override double CalculateArea()
        {
            return Width * Height;
        }

        public override double CalculatePerimeter()
        {
            return 2 * (Width + Height);
        }
    }

    class Program
    {
        static void Main()
        {
            Console.WriteLine("=== การทำงานของ Abstract Class (Shapes) ===\n");

            // Shape s = new Shape("แดง"); // ❌ Compile Error: สร้าง instance จาก abstract class ไม่ได้!

            Shape circle = new Circle("แดงสด", 5.0);
            Shape rect = new Rectangle("น้ำเงิน", 4.0, 6.0);

            Console.WriteLine("⭕ ข้อมูลวงกลม:");
            circle.DisplayInfo();

            Console.WriteLine("🟦 ข้อมูลสี่เหลี่ยมผืนผ้า:");
            rect.DisplayInfo();
        }
    }
}
▶ Output: === การทำงานของ Abstract Class (Shapes) === ⭕ ข้อมูลวงกลม: 🎨 รูปทรงสี: แดงสด 📐 พื้นที่: 78.54 ตร.หน่วย 📏 ความยาวรอบรูป: 31.42 หน่วย 🟦 ข้อมูลสี่เหลี่ยมผืนผ้า: 🎨 รูปทรงสี: น้ำเงิน 📐 พื้นที่: 24.00 ตร.หน่วย 📏 ความยาวรอบรูป: 20.00 หน่วย
07

อีนัมและสมาชิกสแตติก (Enum & Static Members)

กำหนดชุดค่าคงที่ที่มีความหมายชัดเจน และสร้างคลาสเครื่องมือส่วนกลางโดยไม่ต้องสร้างออบเจกต์

⚙️ เครื่องมือส่วนกลางและชุดตัวเลือก (Enum & Static)
📋 ตัวเลือกเมนู และ ตู้เครื่องมือส่วนกลาง

• enum (Enumeration) เหมือน "ตัวเลือกในแบบฟอร์ม" — แทนที่จะเก็บสถานะออเดอร์เป็นข้อความ string "paid", "shipped" ที่อาจพิมพ์ผิดง่าย เรากำหนดเป็น enum OrderStatus.Paid ที่มี Type-Safety และ IntelliSense แนะนำอัตโนมัติ
• static (สมาชิกส่วนกลาง) เหมือน "เครื่องชงกาแฟส่วนกลางของออฟฟิศ" — เป็นของที่แชร์ร่วมกันทั้งบริษัท ทุกคนเรียกใช้ได้ทันทีโดยไม่ต้องซื้อเครื่องชงกาแฟส่วนตัวเข้ามาใหม่ (ไม่ต้องใช้คำสั่ง new)

📄 OrderSystem.cs 
using System;

namespace DotNetLearningHub.OOP
{
    // 1. ประกาศ Enum สถานะของคำสั่งซื้อ
    public enum OrderStatus
    {
        Pending = 1,    // รอดำเนินการ
        Paid = 2,       // ชำระเงินแล้ว
        Shipped = 3,    // จัดส่งแล้ว
        Delivered = 4,  // ส่งถึงผู้รับเรียบร้อย
        Cancelled = 5   // ยกเลิกคำสั่งซื้อ
    }

    // 2. Static Helper Class: คลาสยูทิลิตี้ส่วนกลาง ไม่ต้อง new
    public static class OrderHelper
    {
        public static decimal VatRate = 0.07m; // ภาษีมูลค่าเพิ่ม 7%

        // Static Method คำนวณราคาสุทธิรวม VAT
        public static decimal CalculateGrossTotal(decimal netAmount)
        {
            return netAmount + (netAmount * VatRate);
        }

        // Static Method แปลงสถานะเป็นข้อความภาษาไทยด้วย C# Pattern Matching
        public static string GetStatusBadge(OrderStatus status) => status switch
        {
            OrderStatus.Pending   => "⏳ รอดำเนินการชำระเงิน",
            OrderStatus.Paid      => "💳 ชำระเงินสำเร็จ เตรียมแพ็กของ",
            OrderStatus.Shipped   => "🚚 อยู่ระหว่างจัดส่งสินค้า",
            OrderStatus.Delivered => "✅ สินค้าส่งถึงมือผู้รับแล้ว",
            OrderStatus.Cancelled => "❌ รายการนี้ถูกยกเลิก",
            _                     => "❓ ไม่ทราบสถานะ"
        };
    }

    // 3. คลาสคำสั่งซื้อ
    public class Order
    {
        public string OrderId { get; init; }
        public decimal NetAmount { get; set; }
        public OrderStatus Status { get; set; }

        public Order(string orderId, decimal netAmount)
        {
            OrderId = orderId;
            NetAmount = netAmount;
            Status = OrderStatus.Pending; // เริ่มต้นเป็น Pending เสมอ
        }

        public void PrintInvoice()
        {
            decimal totalWithVat = OrderHelper.CalculateGrossTotal(NetAmount);
            string statusText = OrderHelper.GetStatusBadge(Status);

        }
    }
