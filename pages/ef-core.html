Entity Framework Core (EF Core 8/9) — ORM สำหรับ .NET

เรียนรู้การเชื่อมต่อฐานข้อมูลระดับโปรด้วย Object-Relational Mapping (ORM) มาตรฐานของ .NET ตั้งแต่สถาปัตยกรรม DbContext, Code-First Migrations, CRUD จนถึง Performance Optimization และแก้ปัญหา N+1 Query
🎯 ระดับ: ⭐⭐ กลาง (Intermediate)
⏱️ เวลาเรียน: ~55 นาที
⚡ รางวัล: +100 XP
📦 รองรับ: .NET 8 / .NET 9 LTS
💡
เปรียบเทียบให้เห็นภาพ: EF Core คือ "ล่ามแปลภาษาส่วนตัวระหว่าง C# กับ SQL"
ทำความเข้าใจหัวใจของ Object-Relational Mapper (ORM) ใน 1 นาที
🍽️ เรื่องเล่าเปรียบเทียบ (Analogy)

สมมติว่าคุณเป็นลูกค้าในภัตตาคารนานาชาติที่พูดได้เฉพาะ ภาษาไทย (C# / LINQ) เช่น "ขอสเต็กเนื้อวัวที่ราคาเกิน 100 บาท จัดเรียงจากใหม่สุดมาเก่าสุด 5 จาน"

แต่พ่อครัวในครัวพูดได้เฉพาะ ภาษา SQL (T-SQL, PostgreSQL, MySQL หรือ SQLite) เท่านั้น!
หากไม่มีล่าม คุณจะต้องไปฝึกพูดภาษา SQL โบราณ ต้องจำคำสั่ง เปิดเตาเอง หยิบจานเอง เสี่ยงต่อความผิดพลาด (SQL Injection / Type Mismatch)...

EF Core (Entity Framework Core) ทำหน้าที่เป็น "หัวหน้าบริกรและสุดยอดล่ามแปลภาษา":

  • 🗣️ แปลภาษาอัตโนมัติ: แปลคำสั่ง C# LINQ เช่น .Where(p => p.Price > 100).Take(5) ไปเป็นคำสั่ง SELECT TOP(5) ... WHERE [Price] > 100 ที่ปลอดภัยและมีประสิทธิภาพ
  • 🛡️ ป้องกันอันตราย (SQL Injection): ใช้ Parameterized Query เสมอโดยที่คุณไม่ต้องสั่ง
  • 🍽️ เสิร์ฟใส่จานหรูหรา (Object Mapping): เมื่อครัวส่งข้อมูลดิบกลับมา ล่ามจะจัดใส่จานเป็น C# Object (Class Model) ให้คุณใช้งานได้ทันทีพร้อม IntelliSense
  • 📝 จำสถานะ (Change Tracking): คอยจดจำว่าคุณหยิบช้อนอะไรไปแก้บ้าง พอคุณสั่ง SaveChangesAsync() ล่ามจะส่งบิลไปอัปเดตเฉพาะสิ่งที่เปลี่ยนให้อัตโนมัติ!

โครงสร้างหลักของ EF Core (Core Architecture & Building Blocks)

เข้าใจ 4 องค์ประกอบสำคัญ: DbContext, DbSet<T>, Data Annotations และ Fluent API

ในการทำงานกับ EF Core โค้ดของเราจะประกอบไปด้วย 4 ส่วนหลักที่ทำงานประสานกันอย่างลงตัว:

1. DbContext 🏛️

ตัวแทนเซสชันและการเชื่อมต่อฐานข้อมูล

ทำหน้าที่เป็น Unit of Work และ Repository Pattern ในตัว จัดการ Connection, Transaction, และ Change Tracker ตรวจจับการเปลี่ยนแปลงของ Entity

2. DbSet<TEntity> 📊

ตัวแทนตาราง (Table / Collection)

เปรียบเสมือนตารางใน DB แต่ละตาราง ที่เราสามารถรัน LINQ Query, สั่งเพิ่ม (Add), ค้นหา (Find), แก้ไข (Update) หรือลบ (Remove) ข้อมูลได้ในรูปแบบ C# Collection

3. Entity Classes 📦

โมเดลข้อมูลระดับคลาส (POCO)

Plain Old C# Object ที่แทนแถวข้อมูล (Row) 1 แถวในตาราง โดยมี Properties แทน Columns พร้อมกำหนดกฎความถูกต้องด้วย Data Annotations

4. Fluent API ⚙️

การตั้งค่า Model ขั้นสูงใน OnModelCreating

การกำหนดความสัมพันธ์ซับซ้อน (1-to-1, 1-to-Many, Many-to-Many), Composite Key, Cascade Delete, Index, Table Name, และ Seed Data แยกออกจากตัว Entity ชัดเจน

1.1 การสร้าง Entity Classes และ Data Annotations

มาดูตัวอย่างการออกแบบ Entity สำหรับระบบ E-Commerce ร้านค้าออนไลน์ที่มี Category (หมวดหมู่สินค้า) และ Product (สินค้า):

📄 Models/Product.cs & Models/Category.cs 
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace DotNetShop.Models;

// Entity 1: หมวดหมู่สินค้า
[Table("Categories")] // กำหนดชื่อตารางใน Database
public class Category
{
    [Key] // กำหนดให้เป็น Primary Key (ถ้าชื่อ Id หรือ CategoryId EF จะรู้เองอัตโนมัติ)
    public int Id { get; set; }

    [Required] // บังคับว่าต้องมีค่า (NOT NULL)
    [MaxLength(100)] // กำหนดขนาด VARCHAR(100)
    public string Name { get; set; } = string.Empty;

    public string? Description { get; set; } // Nullable string => NVARCHAR(MAX) NULL

    // Navigation Property: 1 หมวดหมู่ มีได้หลายสินค้า (1-to-Many)
    public ICollection<Product> Products { get; set; } = new List<Product>();
}

// Entity 2: รายการสินค้า
[Table("Products")]
public class Product
{
    [Key]
    public int Id { get; set; }

    [Required]
    [MaxLength(200)]
    public string Name { get; set; } = string.Empty;

    [MaxLength(50)]
    public string Sku { get; set; } = string.Empty;

    [Column(TypeName = "decimal(18,2)")] // กำหนด Data Type ใน SQL แม่นยำ
    [Range(0.01, 1000000.00)]
    public decimal Price { get; set; }

    public int StockQuantity { get; set; }

    public bool IsActive { get; set; } = true;

    public DateTime CreatedAt { get; set; } = DateTime.UtcNow;

    // Foreign Key สำหรับเชื่อมโยงกับ Category
    public int CategoryId { get; set; }

    // Navigation Property: สินค้า 1 ชิ้น อยู่ใน 1 หมวดหมู่
    [ForeignKey(nameof(CategoryId))]
    public Category Category { get; set; } = null!;
}

1.2 การสร้าง DbContext และกำหนดค่าด้วย Fluent API

คลาส ApplicationDbContext จะสืบทอดมาจาก DbContext และบรรจุ DbSet<T> ของทุกตาราง พร้อมกำหนด Fluent API ใน OnModelCreating:

📄 Data/ApplicationDbContext.cs 
using Microsoft.EntityFrameworkCore;
using DotNetShop.Models;

namespace DotNetShop.Data;

public class ApplicationDbContext : DbContext
{
    // Constructor รับ DbContextOptions เพื่อส่ง Connection String เข้ามาจาก Program.cs
    public ApplicationDbContext(DbContextOptions<ApplicationDbContext> options)
        : base(options)
    {
    }

    // DbSet เป็นตัวแทนตาราง Categories และ Products ในฐานข้อมูล
    public DbSet<Category> Categories => Set<Category>();
    public DbSet<Product> Products => Set<Product>();

    // เมธอดสำหรับกำหนด Schema ขั้นสูงด้วย Fluent API
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        // 1. กำหนดความสัมพันธ์ 1-to-Many ระหว่าง Category และ Product
        modelBuilder.Entity<Product>(entity =>
        {
            entity.HasOne(p => p.Category)
                  .WithMany(c => c.Products)
                  .HasForeignKey(p => p.CategoryId)
                  .OnDelete(DeleteBehavior.Restrict); // ป้องกันไม่ให้ลบหมวดหมู่ถ้ายังมีสินค้าอยู่

            // 2. สร้าง Unique Index บน SKU ป้องกันรหัสสินค้าซ้ำ
            entity.HasIndex(p => p.Sku)
                  .IsUnique();

            // 3. กำหนดค่าเริ่มต้นให้กับคอลัมน์ CreatedAt โดยใช้ฟังก์ชันของ SQL Server
            entity.Property(p => p.CreatedAt)
                  .HasDefaultValueSql("GETUTCDATE()");
        });

        // 4. Seeding Data (ข้อมูลตั้งต้น) ลงฐานข้อมูลอัตโนมัติ
        modelBuilder.Entity<Category>().HasData(
            new Category { Id = 1, Name = "Gadgets & Electronics", Description = "อุปกรณ์อิเล็กทรอนิกส์และแกดเจ็ต" },
            new Category { Id = 2, Name = "Books & Media", Description = "หนังสือและสื่อการเรียนรู้" }
        );

        modelBuilder.Entity<Product>().HasData(
            new Product 
            { 
                Id = 1, 
                Name = "Mechanical Keyboard RGB", 
                Sku = "KB-RGB-001", 
                Price = 2490.00m, 
                StockQuantity = 50, 
                CategoryId = 1,
                CreatedAt = new DateTime(2026, 1, 1, 0, 0, 0, DateTimeKind.Utc)
            },
            new Product 
            { 
                Id = 2, 
                Name = "C# in Depth 5th Edition", 
                Sku = "BK-CSHARP-002", 
                Price = 1250.00m, 
                StockQuantity = 30, 
                CategoryId = 2,
                CreatedAt = new DateTime(2026, 1, 1, 0, 0, 0, DateTimeKind.Utc)
            }
        );
    }
}

1.3 การลงทะเบียน DbContext ใน Program.cs (Dependency Injection)

ในแอปพลิเคชัน ASP.NET Core เราจะลงทะเบียน DbContext ผ่าน IoC Container เพื่อให้ Controller หรือ Service เรียกใช้งานผ่าน Dependency Injection ได้แบบ Scoped Lifetime (1 Connection ต่อ 1 HTTP Request):

⚙️ Program.cs & appsettings.json 
// === appsettings.json ===
/*
{
  "ConnectionStrings": {
    "DefaultConnection": "Server=localhost;Database=DotNetShopDb;Trusted_Connection=True;TrustServerCertificate=True;"
  }
}
*/

// === Program.cs ===
using Microsoft.EntityFrameworkCore;
using DotNetShop.Data;

var builder = WebApplication.CreateBuilder(args);

// 1. ดึง Connection String จาก appsettings.json
var connectionString = builder.Configuration.GetConnectionString("DefaultConnection")
    ?? throw new InvalidOperationException("Connection string 'DefaultConnection' not found.");

// 2. ลงทะเบียน ApplicationDbContext ให้ใช้ SQL Server
builder.Services.AddDbContext<ApplicationDbContext>(options =>
    options.UseSqlServer(connectionString, sqlOptions =>
    {
        // เปิดใช้งาน Connection Resiliency (พยายาม Reconnect อัตโนมัติเมื่อหลุด)
        sqlOptions.EnableRetryOnFailure(
            maxRetryCount: 5,
            maxRetryDelay: TimeSpan.FromSeconds(30),
            errorNumbersToAdd: null);
    }));

builder.Services.AddControllers();
var app = builder.Build();
app.MapControllers();
app.Run();
💡
Data Annotations vs Fluent API เลือกใช้อย่างไร?

Data Annotations เหมาะกับกฎพื้นฐาน เช่น [Required] หรือ [MaxLength] เพราะเข้าใจง่ายและแชร์การ Validate ไปยัง UI/API ได้ทันที
ส่วน Fluent API มีพลังสูงสุด รองรับทุกฟีเจอร์ของ DB (เช่น Composite Keys, Complex Index, Shadow Properties, Value Converters, Seed Data) และช่วยให้ Entity Model สะอาด ไม่ผูกติดกับ Database Framework (Separation of Concerns)

02

Code-First Migrations Workflow (วงจรจัดการฐานข้อมูลแบบมือโปร)

จัดการการเปลี่ยนแปลง Schema ของ Database อย่างเป็นระบบด้วยคำสั่ง dotnet ef

Code-First Migrations คือกลไกที่ใช้คลาส C# เป็น Single Source of Truth เมื่อเราแก้ไข C# Model ระบบ EF Core จะเปรียบเทียบความแตกต่าง (Diff) และสร้างไฟล์ Migration ที่ระบุคำสั่ง SQL ในการแก้ไข Database ให้ตรงกับ Model โดยที่ข้อมูลเดิมไม่สูญหาย

PowerShell / Command Prompt — dotnet ef CLI
PS D:\Projects\DotNetShop> dotnet tool install --global dotnet-ef
Tool 'dotnet-ef' (version '8.0.x') was successfully installed.
PS D:\Projects\DotNetShop> dotnet ef migrations add InitialCreate
Build started...
Build succeeded.
Done. To undo this action, use 'dotnet ef migrations remove'
📁 Created file: Migrations/20260827_InitialCreate.cs
PS D:\Projects\DotNetShop> dotnet ef database update
Build started...
Build succeeded.
Applying migration '20260827_InitialCreate'.
✓ Done. Database schema is up to date!

4 คำสั่ง Migrations ที่นักพัฒนา .NET ต้องใช้ทุกวัน

คำสั่ง CLI หน้าที่การทำงาน คำแนะนำ / ข้อควรระวัง
dotnet ef migrations add <Name> สร้างไฟล์ Migration ใหม่โดยเปรียบเทียบ Model C# กับ Snapshot ล่าสุด ตั้งชื่อแบบ PascalCase ชัดเจน เช่น AddProductRating หรือ CreateOrdersTable
dotnet ef database update รัน Migration ที่ยังไม่ได้ Apply ไปยัง Database จริงทันที ใช้ในขั้นตอน Local Development เพื่อทดสอบความถูกต้องของ Schema
dotnet ef migrations remove ยกเลิกและลบไฟล์ Migration ล่าสุดออก (เฉพาะที่ยังไม่รันลง DB) หากรัน database update ไปแล้ว ต้อง rollback ด้วย update <PreviousMigration> ก่อนเสมอ
dotnet ef migrations script -i -o deploy.sql สร้างไฟล์ Raw SQL Script (Idempotent) สำหรับนำไป Deploy ขึ้น Production Best Practice สูงสุด: ใช้ตัวเลือก --idempotent เพื่อให้ script ตรวจสอบตาราง history ก่อนรันเสมอ
🔍
EF Core รู้ได้อย่างไรว่า Migration ไหนรันไปแล้ว?
กลไกเบื้องหลังของตาราง __EFMigrationsHistory

เมื่อเรารันคำสั่ง dotnet ef database update ตัว EF Core จะเข้าไปสร้างและตรวจสอบตารางพิเศษชื่อ __EFMigrationsHistory ภายใน Database ของเราโดยอัตโนมัติ ซึ่งประกอบด้วย 2 คอลัมน์หลัก: 

-- โครงสร้างและข้อมูลในตาราง __EFMigrationsHistory
SELECT MigrationId, ProductVersion FROM [__EFMigrationsHistory];

/* ผลลัพธ์ตัวอย่าง:
MigrationId                         | ProductVersion
------------------------------------+----------------
20260827010000_InitialCreate         | 8.0.4
20260827023015_AddProductSkuIndex   | 8.0.4
*/

EF Core จะนำรายชื่อไฟล์ Migration ทั้งหมดในโปรเจกต์มาเทียบกับรายชื่อในตารางนี้ หากพบไฟล์ใดที่ยังไม่มีชื่อบันทึกอยู่ EF Core จะทยอยรันเฉพาะ Migration ที่ขาดอยู่ตามลำดับเวลา (Timestamp) จนครบ!

⚠️
คำเตือนระดับ Production: ห้ามใช้ context.Database.Migrate() ตอนเปิดแอปเด็ดขาด!

หลายคนมักใส่ context.Database.Migrate() ไว้ใน Program.cs เพื่อให้แอปอัปเดต DB อัตโนมัติเวลาสตาร์ท... แต่วิธีนี้อันตรายมากใน Production เพราะ:
1. Race Condition: เมื่อคุณ deploy แบบ Multi-instance (Kubernetes / App Service หลาย Node) ทุกตัวจะพยายาม Migrate แย่งกัน ทำให้ DB ล็อคหรือพัง
2. Permissions: User ของ Web App ไม่ควรมีสิทธิ์ DDL (CREATE/ALTER TABLE) ใน Production เพื่อความปลอดภัย
แนวทางที่ถูกต้อง: ให้สร้าง SQL Script ด้วย dotnet ef migrations script --idempotent แล้วรันผ่าน CI/CD Pipeline หรือใช้ Migration Bundles (dotnet ef migrations bundle) รันก่อน Deploy แอป

03

การทำ CRUD Operations ด้วย EF Core

Create, Read, Update, Delete ตั้งแต่วิธีมาตรฐานจนถึง High-Performance Bulk Operations ใน EF Core 8/9

3.1 เข้าใจวงจรชีวิตของ Entity (Entity State Machine)

ทุก Entity ที่อยู่ใน DbContext จะถูก Change Tracker ตรวจสอบสถานะเสมอ เมื่อเราเรียก await context.SaveChangesAsync() ตัว EF Core จะดูว่าแต่ละ Object อยู่ในสถานะใด แล้วสร้างคำสั่ง SQL INSERT/UPDATE/DELETE ให้ตรงกัน:

Detached
ไม่อยู่ในการติดตาม

เพิ่งสร้างด้วยคำสั่ง new หรือ query มาด้วย AsNoTracking()

Unchanged
ไม่มีการเปลี่ยนแปลง

โหลดมาจาก Database แล้วข้อมูลยังเหมือนเดิมทุกประการ ไม่สร้างคำสั่ง SQL

Added
เพิ่มใหม่

ผ่านคำสั่ง AddAsync() รอแปลงเป็น INSERT INTO ...

Modified
มีการแก้ไข

Property บางตัวถูกเปลี่ยนค่า รอแปลงเป็น UPDATE ... SET เฉพาะคอลัมน์นั้น

Deleted
ถูกทำเครื่องหมายให้ลบ

ผ่านคำสั่ง Remove() รอแปลงเป็น DELETE FROM ...

3.2 ตัวอย่างโค้ด CRUD ฉบับสมบูรณ์ (Async/Await Best Practices)

ตัวอย่างการนำ EF Core ไปใช้งานใน Service หรือ Controller ของ ASP.NET Core Web API:

📄Create Operations
// 1. เพิ่ม Entity เดี่ยว (Single Item)
var newProduct = new Product
{
    Name = "Wireless Gaming Mouse",
    Sku = "MS-WIR-009",
    Price = 1890.00m,
    StockQuantity = 25,
    CategoryId = 1 // อ้างอิง Category ID ที่มีอยู่จริง
};

// นำเข้าสู่ Change Tracker ในสถานะ EntityState.Added
await context.Products.AddAsync(newProduct);

// บันทึกการเปลี่ยนแปลงลงฐานข้อมูล (รันคำสั่ง SQL INSERT)
await context.SaveChangesAsync();

// หลัง SaveChangesAsync() สำเร็จ EF Core จะดึง Generated ID จาก Database มาใส่ในตัวแปรทันที!
Console.WriteLine($"สร้างสินค้าสำเร็จ! รหัสสินค้าใหม่คือ ID: {newProduct.Id}");

// ----------------------------------------------------
// 2. เพิ่มข้อมูลเป็นชุด (Batch Insert)
var bulkItems = new List<Product>
{
    new() { Name = "Mouse Pad XXL", Sku = "PAD-XXL-01", Price = 490m, CategoryId = 1 },
    new() { Name = "USB-C Hub 7-in-1", Sku = "HUB-7IN1-02", Price = 990m, CategoryId = 1 }

await context.Products.AddRangeAsync(bulkItems);
await context.SaveChangesAsync();
📄Read & LINQ Query Operations
// 1. ค้นหาด้วย Primary Key โดยตรงด้วย FindAsync
// (ข้อดี: FindAsync จะเช็คในหน่วยความจำของ DbContext ก่อน ถ้ายืมมาแล้วจะไม่ยิง DB ซ้ำ)
Product? product = await context.Products.FindAsync(1);

// 2. ค้นหาข้อมูลตัวแรกตามเงื่อนไข (FirstOrDefaultAsync)
Product? activeItem = await context.Products
    .FirstOrDefaultAsync(p => p.Sku == "KB-RGB-001");

// 3. กรองข้อมูล (Filter) + จัดเรียง (Sort) + ดึงเป็น List
List<Product> expensiveItems = await context.Products
    .Where(p => p.IsActive && p.Price >= 1000m)
    .OrderByDescending(p => p.Price)
    .ToListAsync();

// 4. การนับจำนวนและตรวจสอบการมีอยู่ (รันคำสั่ง COUNT และ EXISTS บน DB โดยตรง ไม่โหลด data)
int totalActiveCount = await context.Products.CountAsync(p => p.IsActive);
bool hasOutOfStock = await context.Products.AnyAsync(p => p.StockQuantity == 0);

// 5. การแบ่งหน้า (Pagination) ด้วย Skip & Take
int pageNumber = 1;
int pageSize = 10;
var pagedProducts = await context.Products
    .OrderBy(p => p.Id)
    .Skip((pageNumber - 1) * pageSize)
    .Take(pageSize)
    .ToListAsync();
📄Update Operations
// วิธีที่ 1: Connected Update (แนะนำที่สุดสำหรับ Business Logic)
// ดึงข้อมูลมาก่อน -> แก้ไข Property -> สั่ง SaveChangesAsync
var product = await context.Products.FindAsync(1);
if (product != null)
{
    product.Price = 2190.00m;       // ปรับลดราคา
    product.StockQuantity -= 1;      // หักสต็อก
    
    // Change Tracker จะตรวจจับอัตโนมัติว่าฟิลด์ Price กับ StockQuantity เปลี่ยนไป
    // และสร้าง SQL: UPDATE Products SET Price = 2190, StockQuantity = 49 WHERE Id = 1
    await context.SaveChangesAsync();
}

// วิธีที่ 2: Disconnected Update (เช่น ได้รับ DTO มาจาก Web API โดยไม่ได้โหลดจาก DB ก่อน)
var updatedProduct = new Product
{

    Name = "Mechanical Keyboard Pro RGB",
    Sku = "KB-RGB-001",

context.Products.Update(updatedProduct);
await context.SaveChangesAsync();
📄Delete & Soft Delete Pattern
// 1. การลบข้อมูลจริง (Hard Delete)
var productToDelete = await context.Products.FindAsync(2);
if (productToDelete != null)
{
    // เปลี่ยนสถานะ Entity เป็น EntityState.Deleted
    context.Products.Remove(productToDelete);
    await context.SaveChangesAsync(); // รันคำสั่ง DELETE FROM Products WHERE Id = 2
}

// 2. Soft Delete Pattern (วิธีมาตรฐานในองค์กร: ซ่อนข้อมูลโดยไม่ลบจริง)
// คลาสมี Property: public bool IsDeleted { get; set; }

var item = await context.Products.FindAsync(1);
if (item != null)
{
    item.IsDeleted = true; // ทำเครื่องหมายว่าถูกลบ
    await context.SaveChangesAsync();
}

// modelBuilder.Entity<Product>().HasQueryFilter(p => !p.IsDeleted);
⚡ExecuteUpdateAsync & ExecuteDeleteAsync (EF Core 7/8/9)
// ⚡ ฟีเจอร์ใหม่ความเร็วแสง: สั่งรันคำสั่งตรงบน Database ทันที 1 Query
// โดย "ไม่ต้องโหลด Entity ลงหน่วยความจำ" และ "ไม่ต้องผ่าน Change Tracker"!

// 1. Bulk Update: ปรับลดราคาสินค้า 10% ทุกรายการในหมวดหมู่ Gadgets (CategoryId = 1)
int affectedRows = await context.Products
    .Where(p => p.CategoryId == 1 && p.IsActive)
    .ExecuteUpdateAsync(setters => setters
        .SetProperty(p => p.Price, p => p.Price * 0.90m)
        .SetProperty(p => p.StockQuantity, p => p.StockQuantity + 5)
    );

Console.WriteLine($"อัปเดตราคาสำเร็จทั้งหมด {affectedRows} รายการ!");
// สร้าง SQL ทันที: UPDATE [p] SET [p].[Price] = [p].[Price] * 0.9, [p].[StockQuantity] = [p].[StockQuantity] + 5 FROM [Products] AS [p] WHERE ...

// 2. Bulk Delete: ลบสินค้าที่ยกเลิกการขายและหมดสต็อกทั้งหมดใน 1 คำสั่ง
int deletedRows = await context.Products
    .Where(p => !p.IsActive && p.StockQuantity == 0)
    .ExecuteDeleteAsync();

Console.WriteLine($"ลบสินค้าเก่าสำเร็จ {deletedRows} รายการ!");
// สร้าง SQL ทันที: DELETE FROM [Products] WHERE [IsActive] = 0 AND [StockQuantity] = 0

04

Performance Best Practices ที่ Junior .NET ต้องรู้

เทคนิคระดับ Senior ในการรีดความเร็ว EF Core, กำจัดคอขวด, และแก้ปัญหา N+1 Query Problem

4.1 AsNoTracking() สำหรับ Read-Only Queries

ตามปกติเมื่อเรา Query ข้อมูลจาก EF Core ตัว Change Tracker จะทำการจดจำ Snapshot ของ Entity ทุกตัวไว้ในหน่วยความจำ RAM เพื่อรอตรวจจับว่ามีการแก้ไขค่าหรือไม่
หากเราต้องการอ่านข้อมูลมาแสดงผลเท่านั้น (เช่น ทำ API ส่ง JSON กลับไปให้ Frontend) ให้ใส่ .AsNoTracking() เสมอ:

❌ แบบช้าและเปลือง RAM (Tracking ทำงาน)
// Change Tracker จะโคลน Object เก็บไว้ใน Snapshot Memory
var products = await context.Products
    .Where(p => p.Price > 500)
    .ToListAsync();
// เหมาะสำหรับตอนที่ "ต้องการแก้ค่าแล้วสั่ง SaveChangesAsync" เท่านั้น
✅ เร็วกว่า 2-4 เท่า และประหยัด RAM มหาศาล
// ปิด Change Tracker ทำให้ประหยัด RAM และ Query เร็วขึ้นสูงสุด
var products = await context.Products
    .AsNoTracking()
    .Where(p => p.Price > 500)
    .ToListAsync();
// แนะนำเป็นค่า Default สำหรับ GET Endpoint ทั้งหมดใน Web API!

4.2 หายนะ N+1 Query Problem และการแก้ด้วย Eager Loading

N+1 Problem คือข้อผิดพลาดคลาสสิกที่ทำให้ Database Server ล่มได้ง่ายที่สุด เกิดจากการโหลดข้อมูลหลัก 1 ครั้ง (1 Query) แล้วเขียนลูปไปดึงข้อมูลตารางลูกอีก N ครั้ง (N Queries):

💥 ตัวอย่างโค้ดที่ทำให้เกิด N+1 Query (อย่าหาทำ!):

// 1. ยิง Query แรกเพื่อดึง Category 100 รายการ (1 Query)
var categories = await context.Categories.ToListAsync();

foreach (var cat in categories)
{
    // 2. เผลอยิง Query ย่อยใน Loop เพื่อดึงสินค้าของแต่ละหมวดหมู่อีก 100 ครั้ง! (N Queries)
    var products = await context.Products.Where(p => p.CategoryId == cat.Id).ToListAsync();
    Console.WriteLine($"{cat.Name}: มีสินค้า {products.Count} ชิ้น");
}
// ผลลัพธ์: ยิง SQL ทั้งหมด 1 + 100 = 101 ครั้งลง Database พร้อมกัน! ระบบค้างแน่นอน!

วิธีแก้ด้วย 3 กลยุทธ์การโหลดข้อมูล (Loading Strategies):

ใช้คำสั่ง .Include() และ .ThenInclude() เพื่อสั่งให้ EF Core เขียนคำสั่ง SQL LEFT JOIN นำข้อมูลตารางที่เกี่ยวข้องกันทั้งหมดขึ้นมาตั้งแต่คำสั่งแรกเพียง Query เดียว:

// Eager Loading: ยิง SQL JOIN ครั้งเดียวจบ ไม่เกิด N+1!
var categoriesWithProducts = await context.Categories
    .AsNoTracking()
    .Include(c => c.Products) // โหลดตารางลูก Products
    .ToListAsync();

// หากมีความสัมพันธ์หลายชั้น เช่น Category -> Products -> Reviews
var fullDetails = await context.Categories
    .AsNoTracking()
    .Include(c => c.Products)
        .ThenInclude(p => p.Reviews) // โหลดตารางหลานต่อจาก Products
    .ToListAsync();

โหลดข้อมูล Entity หลักมาก่อน แล้วค่อยเลือกโหลดข้อมูลตารางลูกเฉพาะบางเรคคอร์ดที่ตรงตามเงื่อนไขในภายหลังผ่าน Entry():

var category = await context.Categories.FindAsync(1);

if (category != null && needProducts)
{
    // โหลด Products เฉพาะเมื่อต้องการใช้จริง และสามารถกรองเงื่อนไขต่อได้
    await context.Entry(category)
        .Collection(c => c.Products)
        .Query()
        .Where(p => p.IsActive && p.Price > 500)
        .LoadAsync();
}

โหลดข้อมูลตารางลูกอัตโนมัติเมื่อมีการเรียกใช้ Property (ต้องเปิด Proxies และใช้ virtual):

// ต้องลง NuGet: Microsoft.EntityFrameworkCore.Proxies
// และเปิด options.UseLazyLoadingProxies() ใน Program.cs
public class Category
{
    public int Id { get; set; }
    public string Name { get; set; } = string.Empty;
    public virtual ICollection<Product> Products { get; set; } = new List<Product>();
}

// ⚠️ อันตรายมากใน Web API! เพราะเมื่อ JSON Serializer ทำงาน มันจะพยายาม serialize
// navigation property ทุกตัว ทำให้เกิดการยิง Query อัตโนมัตินับพันครั้งและเกิด Circular Reference!

4.3 การแก้ Cartesian Explosion ด้วย Split Queries (.AsSplitQuery())

เมื่อเราใช้ .Include() ตารางลูกหลายๆ ตารางพร้อมกัน (เช่น Order มีหลาย OrderItems และหลาย OrderLogs) การทำ SQL Single Query ด้วย LEFT JOIN หลายตารางจะทำให้ข้อมูลผลลัพธ์คูณทวีคูณ (Cartesian Explosion) เช่น ได้ข้อมูลซ้ำซ้อน 10,000 แถวกลับมา ทั้งที่มี Order แค่ 10 ใบ!
ใน EF Core เราสามารถสั่ง .AsSplitQuery() เพื่อให้ระบบแยกคำสั่ง SQL ยิงทีละตารางอย่างมีประสิทธิภาพ:

⚡Split Query Optimization
// แยกเป็น 2-3 คำสั่ง SQL ย่อย แทนที่จะ JOIN เป็นก้อนยักษ์
var categories = await context.Categories
    .AsNoTracking()
    .Include(c => c.Products)
    .Include(c => c.Promotions)
    .AsSplitQuery() // 👈 แบ่งคำสั่ง SQL แยกกันเพื่อป้องกันข้อมูลซ้ำซ้อน
    .ToListAsync();

// หรือกำหนดให้ทุก Query เป็น Split Query เป็นค่า Default ใน Program.cs:
// options.UseSqlServer(connectionString, o => o.UseQuerySplittingBehavior(QuerySplittingBehavior.SplitQuery));

4.4 Projection ด้วย .Select() (ดึงเฉพาะฟิลด์ที่ต้องใช้)

แทนที่จะดึง Entity มาทั้งตาราง (ซึ่งมีคอลัมน์รูปภาพ, Description ยาวๆ, วันที่สร้าง) ให้ดึงเฉพาะฟิลด์ที่จำเป็นเข้าสู่ DTO (Data Transfer Object) ซึ่ง EF Core จะแปลงเป็น SELECT Name, Price FROM ... แทน SELECT * ช่วยประหยัด Network I/O และ RAM ได้ดีที่สุด:

🚀DTO Projection Pattern
// DTO Class สำหรับส่งข้อมูลให้ Mobile App / Web Frontend
public record ProductSummaryDto(int Id, string Name, decimal Price, string CategoryName);

// การ Query ด้วย Projection (ไม่จำเป็นต้องใส่ AsNoTracking() เพราะ Projection ไม่ถูก track อยู่แล้ว)
List<ProductSummaryDto> dtos = await context.Products
    .Where(p => p.IsActive)
    .Select(p => new ProductSummaryDto(
        p.Id,
        p.Name,
        p.Price,
        p.Category.Name // EF Core จะเขียน JOIN ดึงชื่อหมวดหมู่อัตโนมัติใน SELECT เดียว!
    ))
    .ToListAsync();
